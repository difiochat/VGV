import { createClient } from "npm:@supabase/supabase-js@2"; import * as XLSX from "npm:xlsx@0.18.5"; const ADMIN_UID = "2b424b51-49a8-4d66-a5a2-45fc3fcd1075"; const corsHeaders = { "Access-Control-Allow-Origin": "*", "Access-Control-Allow-Headers": "authorization, apikey, content-type, x-client-info", "Access-Control-Allow-Methods": "POST, OPTIONS", }; function normalize(value: unknown): string { return String(value ?? "") .trim() .toLowerCase() .normalize("NFD") .replace(/[\u0300-\u036f]/g, ""); } Deno.serve(async (req) => { const respond = (data: unknown, status = 200) => new Response(JSON.stringify(data), { status, headers: { ...corsHeaders, "Content-Type": "application/json", }, }); if (req.method === "OPTIONS") { return new Response("ok", { headers: corsHeaders }); } if (req.method !== "POST") { return respond( { error: "Usa POST para cargar el Excel." }, 405 ); } const authorization = req.headers.get("Authorization"); if (!authorization?.startsWith("Bearer ")) { return respond({ error: "Debes iniciar sesión." }, 401); } try { const url = Deno.env.get("SUPABASE_URL"); const publishableRaw = Deno.env.get( "SUPABASE_PUBLISHABLE_KEYS" ); const secretRaw = Deno.env.get("SUPABASE_SECRET_KEYS"); if (!url || !publishableRaw || !secretRaw) { return respond( { error: "Falta configurar la URL o las claves del servidor.", }, 500 ); } const publishableKeys = JSON.parse(publishableRaw); const secretKeys = JSON.parse(secretRaw); const anonKey = publishableKeys["default"]; const serviceKey = secretKeys["default"]; if (!anonKey || !serviceKey) { return respond( { error: "No se encontraron las claves predeterminadas de Supabase.", }, 500 ); } // Verificar que el usuario sea el administrador autorizado. const token = authorization.slice(7); const authClient = createClient(url, anonKey, { auth: { persistSession: false, autoRefreshToken: false, }, }); const { data: authData, error: authError } = await authClient.auth.getUser(token); if (authError || authData.user?.id !== ADMIN_UID) { return respond( { error: "No tienes permiso de administrador." }, 403 ); } // Recibir el archivo Excel. const form = await req.formData(); const file = form.get("file"); if (!(file instanceof File)) { return respond( { error: "Selecciona un archivo Excel." }, 400 ); } if (!/\.(xlsx|xls)$/i.test(file.name)) { return respond( { error: "El archivo debe ser .xlsx o .xls." }, 400 ); } // Leer la primera hoja del Excel. const bytes = new Uint8Array(await file.arrayBuffer()); const workbook = XLSX.read(bytes, { type: "array" }); const firstSheet = workbook.SheetNames[0]; const sheet = workbook.Sheets[firstSheet]; } // Identificar las columnas aunque tengan tildes o // diferencias entre mayúsculas y minúsculas. const headers = Object.keys(rows[0]); const findColumn = (names: string[]) => headers.find((header) => names.includes(normalize(header)) ); const codeCol = findColumn(["codigo"]); const productCol = findColumn([ "nombre producto", "producto", ]); const stockCol = findColumn(["stock"]); const priceCol = findColumn(["pvp5"]); if (!codeCol || !productCol || !stockCol || !priceCol) { return respond( { error: "Faltan columnas necesarias. Se requieren Codigo, Nombre Producto, Stock y Pvp5.", columnasEncontradas: headers, }, 400 ); } const products: { codigo: string; producto: string; existencias: number; precio_pvp5: number; actualizado_en: string; }[] = []; const seenCodes = new Set(); for (const row of rows) { const rawCode = row[codeCol]; const rawName = row[productCol]; const rawStock = row[stockCol]; const rawPrice = row[priceCol]; // Ignorar filas completamente vacías. if ( [rawCode, rawName, rawStock, rawPrice].every( (value) => value === null || value === undefined || String(value).trim() === "" ) ) { continue; } const codigo = String(rawCode ?? "").trim(); const producto = String(rawName ?? "").trim(); if (!codigo || !producto) { return respond( { error: "Hay una fila con Codigo o Nombre Producto vacío.", }, 400 ); } if (seenCodes.has(codigo)) { return respond( { error: `El código ${codigo} está duplicado en el Excel.`, }, 400 ); } seenCodes.add(codigo); const existencias = Number(rawStock); const precio_pvp5 = Number(rawPrice); if ( rawStock === null || rawStock === undefined || String(rawStock).trim() === "" || !Number.isInteger(existencias) || existencias < 0 ) { return respond( { error: `Stock inválido para ${producto} (código ${codigo}).`, }, 400 ); } if ( rawPrice === null || rawPrice === undefined || String(rawPrice).trim() === "" || !Number.isFinite(precio_pvp5) || precio_pvp5 < 0 ) { return respond( { error: `Pvp5 inválido para ${producto} (código ${codigo}).`, }, 400 ); } products.push({ codigo, producto, existencias, precio_pvp5, actualizado_en: new Date().toISOString(), }); } if (!products.length) { return respond( { error: "No se encontraron productos válidos." }, 400 ); } // Conectar con Supabase usando la clave secreta // exclusivamente en el servidor. const db = createClient(url, serviceKey, { auth: { persistSession: false, autoRefreshToken: false, }, }); // Reemplazar el inventario mediante la función SQL. const { data: cantidad, error } = await db.rpc( "reemplazar_inventario_vgv", { p_productos: products } ); if (error) { console.error("Error al reemplazar inventario:", error); return respond( { error: "No se pudo reemplazar el inventario.", detalle: error.message, }, 500 ); } return respond({ mensaje: "Inventario actualizado correctamente.", productosProcesados: products.length, productosActualizados: cantidad, fuentePrecio: "Pvp5", identificador: "Codigo", }); } catch (error) { console.error("Error al procesar Excel:", error); return respond( { error: "No se pudo procesar el Excel.", detalle: error instanceof Error ? error.message : String(error), }, 400 ); } });
